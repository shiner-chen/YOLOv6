```python
        elif self.iou_type == 'wiou':
            # Wise-IoU: exp(rho^2/C^2) as detached gradient weight
            # focuses training on hard (low-quality) examples
            c2 = cw ** 2 + ch ** 2 + self.eps
            rho2 = ((b2_x1 + b2_x2 - b1_x1 - b1_x2) ** 2 +
                    (b2_y1 + b2_y2 - b1_y1 - b1_y2) ** 2) / 4
            wise_factor = torch.exp(rho2 / c2)
            loss = (1.0 - iou) * wise_factor.detach()
            if self.reduction == 'sum':
                return loss.sum()
            elif self.reduction == 'mean':
                return loss.mean()
            return loss
```

```python
# yolov6/assigners/stal_assigner.py
"""STAL: Small Target Aware Label Assignment.

Extends TaskAlignedAssigner with a small-target boost factor applied to the
alignment metric, so that tiny GT boxes compete more effectively for anchors.

Reference: SDD-YOLO (2024), Section III-B.
"""
import torch
from yolov6.assigners.tal_assigner import TaskAlignedAssigner


class STALAssigner(TaskAlignedAssigner):
    """TAL with small-target aware alignment metric boost.

    For each GT box, computes a boost factor:
        small_factor = 1 + gamma * exp(-area_ratio / area_thr)
    where area_ratio = (gt_w * gt_h) / ori_img_size^2.

    For large boxes  area_ratio >> area_thr → exp ≈ 0 → factor ≈ 1 (no boost).
    For small boxes  area_ratio << area_thr → exp ≈ 1 → factor ≈ 1 + gamma.

    Args:
        topk: top-k candidates per GT (use 13 for O2M, 1 for O2O).
        num_classes: number of object classes.
        alpha, beta: exponents in align_metric = score^alpha * iou^beta.
        gamma: magnitude of small-target boost (default 0.5).
        area_thr: normalised area threshold; GTs below this get max boost
                  (default 0.02, i.e. 2 % of image area).
        ori_img_size: image side length used for normalisation (default 320).
    """

    def __init__(self, topk=13, num_classes=80,
                 alpha=1.0, beta=6.0, eps=1e-9,
                 gamma=0.5, area_thr=0.02, ori_img_size=320):
        super().__init__(topk=topk, num_classes=num_classes,
                         alpha=alpha, beta=beta, eps=eps)
        self.gamma = gamma
        self.area_thr = area_thr
        self.img_area = float(ori_img_size) ** 2

    def get_pos_mask(self, pd_scores, pd_bboxes, gt_labels, gt_bboxes,
                     anc_points, mask_gt):
        """Override parent to inject STAL boost into align_metric."""
        from yolov6.assigners.assigner_utils import (
            select_candidates_in_gts, select_highest_overlaps)

        # Standard alignment metric (bs, n_max_boxes, n_anchors)
        align_metric, overlaps = self.get_box_metrics(
            pd_scores, pd_bboxes, gt_labels, gt_bboxes)

        # ---- small-target boost ----
        # gt_bboxes: (bs, n_max_boxes, 4) in pixel xyxy
        gt_w = (gt_bboxes[..., 2] - gt_bboxes[..., 0]).clamp(min=0)
        gt_h = (gt_bboxes[..., 3] - gt_bboxes[..., 1]).clamp(min=0)
        area_ratio = (gt_w * gt_h) / self.img_area          # (bs, n_max_boxes)
        small_factor = 1.0 + self.gamma * torch.exp(
            -area_ratio / self.area_thr)                     # (bs, n_max_boxes)
        align_metric = align_metric * small_factor.unsqueeze(-1)  # broadcast

        # Rest is identical to TaskAlignedAssigner.get_pos_mask
        mask_in_gts = select_candidates_in_gts(anc_points, gt_bboxes)
        mask_topk = self.select_topk_candidates(
            align_metric * mask_in_gts,
            topk_mask=mask_gt.repeat([1, 1, self.topk]).bool())
        mask_pos = mask_topk * mask_in_gts * mask_gt

        return mask_pos, align_metric, overlaps
```

- [ ] **Step 3：创建 effidehead_o2o.py（前50行：imports + build_effidehead_layer）**

由于文件较长，分两步创建。先写 `build_effidehead_layer` 函数（为4-scale P2 head 创建7×4=28个层：每个 scale 含 stem, cls_conv, reg_conv, cls_pred_o2m, reg_pred_o2m, cls_pred_o2o, reg_pred_o2o）：

```python
# yolov6/models/heads/effidehead_o2o.py
"""Dual-branch O2M/O2O detection head with shared stem/cls_conv/reg_conv.

Training: returns (feats, cls_o2m, reg_o2m, cls_o2o, reg_o2o).
Inference: only O2O branch runs, returns (B, N_all, 4+1+nc).
"""
import torch
import torch.nn as nn
import torch.nn.functional as F
import math
from yolov6.layers.common import ConvBNSiLU
from yolov6.assigners.anchor_generator import generate_anchors
from yolov6.utils.general import dist2bbox


class Detect(nn.Module):
    export = False

    def __init__(self, num_classes=80, num_layers=4, inplace=True,
                 head_layers=None, use_dfl=False, reg_max=0):
        super().__init__()
        assert head_layers is not None
        assert use_dfl is False and reg_max == 0, "O2O head: DFL must be off"

        self.nc = num_classes
        self.no = num_classes + 5
        self.nl = num_layers
        self.grid = [torch.zeros(1)] * num_layers
        self.prior_prob = 1e-2
        self.inplace = inplace
        self.stride = torch.tensor([4, 8, 16, 32])  # P2/P3/P4/P5
        self.grid_cell_offset = 0.5
        self.grid_cell_size = 5.0

        # Init decouple head: shared stem/cls_conv/reg_conv, separate o2m/o2o pred
        self.stems = nn.ModuleList()
        self.cls_convs = nn.ModuleList()
        self.reg_convs = nn.ModuleList()
        self.cls_preds_o2m = nn.ModuleList()
        self.reg_preds_o2m = nn.ModuleList()
        self.cls_preds_o2o = nn.ModuleList()
        self.reg_preds_o2o = nn.ModuleList()

        for i in range(num_layers):
            idx = i * 7
            self.stems.append(head_layers[idx])
            self.cls_convs.append(head_layers[idx+1])
            self.reg_convs.append(head_layers[idx+2])
            self.cls_preds_o2m.append(head_layers[idx+3])
            self.reg_preds_o2m.append(head_layers[idx+4])
            self.cls_preds_o2o.append(head_layers[idx+5])
            self.reg_preds_o2o.append(head_layers[idx+6])

    def initialize_biases(self):
        """Initialize prediction head biases to prior_prob."""
        for conv_list in [self.cls_preds_o2m, self.cls_preds_o2o]:
            for conv in conv_list:
                b = conv.bias.view(-1)
                b.data.fill_(-math.log((1 - self.prior_prob) / self.prior_prob))
                conv.bias = nn.Parameter(b.view(-1), requires_grad=True)
                conv.weight.data.fill_(0.)

        for conv_list in [self.reg_preds_o2m, self.reg_preds_o2o]:
            for conv in conv_list:
                b = conv.bias.view(-1)
                b.data.fill_(1.0)
                conv.bias = nn.Parameter(b.view(-1), requires_grad=True)
                conv.weight.data.fill_(0.)

    def forward(self, x):
        """
        x: list of 4 feature maps [P2, P3, P4, P5]
        Training: returns (feats, cls_o2m, reg_o2m, cls_o2o, reg_o2o)
        Inference: returns (B, N_all, 4+1+nc)
        """
        if self.training:
            device = x[0].device
            cls_score_list_o2m = []
            reg_dist_list_o2m = []
            cls_score_list_o2o = []
            reg_dist_list_o2o = []

            for i in range(self.nl):
                b, _, h, w = x[i].shape
                feat = self.stems[i](x[i])
                cls_feat = self.cls_convs[i](feat)
                reg_feat = self.reg_convs[i](feat)

                # O2M branch
                cls_output_o2m = self.cls_preds_o2m[i](cls_feat)
                reg_output_o2m = self.reg_preds_o2m[i](reg_feat)
                cls_score_list_o2m.append(torch.sigmoid(cls_output_o2m).flatten(2).permute(0, 2, 1))
                reg_dist_list_o2m.append(reg_output_o2m.flatten(2).permute(0, 2, 1))

                # O2O branch
                cls_output_o2o = self.cls_preds_o2o[i](cls_feat)
                reg_output_o2o = self.reg_preds_o2o[i](reg_feat)
                cls_score_list_o2o.append(torch.sigmoid(cls_output_o2o).flatten(2).permute(0, 2, 1))
                reg_dist_list_o2o.append(reg_output_o2o.flatten(2).permute(0, 2, 1))

            cls_o2m = torch.cat(cls_score_list_o2m, dim=1)
            reg_o2m = torch.cat(reg_dist_list_o2m, dim=1)
            cls_o2o = torch.cat(cls_score_list_o2o, dim=1)
            reg_o2o = torch.cat(reg_dist_list_o2o, dim=1)

            return x, cls_o2m, reg_o2m, cls_o2o, reg_o2o

        else:
            # Inference: only O2O branch
            device = x[0].device
            cls_score_list = []
            reg_dist_list = []

            for i in range(self.nl):
                b, _, h, w = x[i].shape
                feat = self.stems[i](x[i])
                cls_feat = self.cls_convs[i](feat)
                reg_feat = self.reg_convs[i](feat)

                cls_output = self.cls_preds_o2o[i](cls_feat)
                reg_output = self.reg_preds_o2o[i](reg_feat)

                cls_score_list.append(torch.sigmoid(cls_output).reshape([b, self.nc, h*w]))
                reg_dist_list.append(reg_output.reshape([b, 4, h*w]))

            cls_score_list = torch.cat(cls_score_list, dim=-1).permute(0, 2, 1)
            reg_dist_list = torch.cat(reg_dist_list, dim=-1).permute(0, 2, 1)

            anchor_points, stride_tensor = generate_anchors(
                x, self.stride, self.grid_cell_size, self.grid_cell_offset,
                device=device, is_eval=True, mode='af')

            pred_bboxes = dist2bbox(reg_dist_list, anchor_points, box_format='xywh')
            pred_bboxes *= stride_tensor

            return torch.cat([
                pred_bboxes,
                torch.ones((b, pred_bboxes.shape[1], 1), device=device, dtype=pred_bboxes.dtype),
                cls_score_list
            ], dim=-1)


def build_effidehead_layer(channels_list, num_classes, num_layers=4):
    """Build 7 layers per scale: stem, cls_conv, reg_conv, cls_pred_o2m, reg_pred_o2m, cls_pred_o2o, reg_pred_o2o."""
    chx = [12, 6, 8, 10]  # P2(/4), P3(/8), P4(/16), P5(/32)

    layers = []
    for i in range(num_layers):
        ch = channels_list[chx[i]]
        layers.extend([
            # stem
            ConvBNSiLU(ch, ch, kernel_size=1, stride=1),
            # cls_conv
            ConvBNSiLU(ch, ch, kernel_size=3, stride=1),
            # reg_conv
            ConvBNSiLU(ch, ch, kernel_size=3, stride=1),
            # cls_pred_o2m
            nn.Conv2d(ch, num_classes, kernel_size=1),
            # reg_pred_o2m
            nn.Conv2d(ch, 4, kernel_size=1),
            # cls_pred_o2o
            nn.Conv2d(ch, num_classes, kernel_size=1),
            # reg_pred_o2o
            nn.Conv2d(ch, 4, kernel_size=1),
        ])

    return nn.Sequential(*layers)
```

   ```python
   class ConfidenceMarginLoss(nn.Module):
       def __init__(self, threshold=0.25, n_steps=3):
           super().__init__()
           self.margin = n_steps / 255.0
           self.hi = threshold + self.margin
           self.lo = threshold - self.margin

       def forward(self, cls_scores, fg_mask):
           pos = cls_scores[fg_mask]
           neg = cls_scores[~fg_mask]
           loss_pos = torch.clamp(self.hi - pos, min=0).mean()
           loss_neg = torch.clamp(neg - self.lo, min=0).mean()
           return loss_pos + loss_neg
   ```
